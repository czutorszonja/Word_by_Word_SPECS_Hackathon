// TEMPORARY verification probe — drives the flow to a given stage for screenshots / checks.
// Remove before finishing the build.
import {WbwMain} from "./WbwMain"
import {Button} from "SpectaclesUIKit.lspkg/Scripts/Components/Button/Button"
import {camPos, camLook, placeFacingUser} from "./WbwSpatial"

@component
export class WbwTempProbe extends BaseScriptComponent {
  @input main!: WbwMain
  @input @hint("setup | wordcard | phrases | phrases2 | practice | flow") stage: string = "setup"

  onAwake(): void {
    const ev = this.createEvent("DelayedCallbackEvent") as DelayedCallbackEvent
    ev.bind(() => this.run())
    ev.reset(1.5)
  }

  private later(t: number, f: () => void): void {
    const ev = this.createEvent("DelayedCallbackEvent") as DelayedCallbackEvent
    ev.bind(f)
    ev.reset(t)
  }

  private find(so: SceneObject, name: string): SceneObject | null {
    if (so.name === name) return so
    for (let i = 0; i < so.getChildrenCount(); i++) {
      const r = this.find(so.getChild(i), name)
      if (r) return r
    }
    return null
  }

  private frame(targetName: string, anchorName?: string): void {
    const cafe = (this.main as any).cafe.getSceneObject() as SceneObject
    const ct = cafe.getTransform()
    const target = this.find(cafe, targetName)
    const anchor = anchorName ? this.find(cafe, anchorName) : null
    if (!target) { print(`PROBE view: no ${targetName}`); return }
    const ap = anchor ? anchor.getTransform().getWorldPosition() : ct.getWorldPosition()
    const v = target.getTransform().getWorldPosition().sub(ap)
    const look = camLook()
    const cur = ct.getWorldRotation()
    const dYaw = Math.atan2(-look.x, -look.z) - Math.atan2(-v.x, -v.z)
    const rot = quat.fromEulerAngles(0, dYaw, 0)
    // rotate the café about the anchor, then move the anchor under the head
    const p = ct.getWorldPosition()
    const rel = rot.multiplyVec3(p.sub(ap))
    ct.setWorldRotation(rot.multiply(cur))
    const c = camPos()
    ct.setWorldPosition(new vec3(c.x + rel.x, p.y + rel.y, c.z + rel.z))
    print(`PROBE view ${targetName} from ${anchorName || "start"} yaw ${(dYaw * 180 / Math.PI).toFixed(0)}`)
  }

  // Full done-criteria walk-through: setup → toast → menu → Learn → Use it live (7 steps) → Practice.
  private runFlow(): void {
    const m = this.main as any
    const L = m.learn, O = m.order, P = m.practice
    const R: string[] = []
    const ok = (name: string, cond: boolean, extra: string = "") => { R.push(`${cond ? "PASS" : "FAIL"} ${name} ${extra}`); print(`PROBE ${cond ? "PASS" : "FAIL"} ${name} ${extra}`) }
    // C1 setup → toast → mode menu
    ok("C1.setupVisibleAtStart", !m.setupComplete && m.setupPanel.isVisible !== false)
    m.confirmSetup()
    ok("C1.setupComplete", m.setupComplete === true)
    ok("C1.toastShown", m.toast.isVisible === true)
    this.later(0.5, () => {
      m.toggleMenu()
      ok("C1.menuOpen", m.menuOpen === true)
      m.selectMode("learn")
      ok("C1.menuClosedAfterPick", m.menuOpen === false, `mode=${m.currentMode}`)
      this.frame("SandwichCounter")
      // C2 Learn
      this.later(2.0, () => {
        ok("C2.bracketsShown", L.bracketedIds.length >= 1 && L.bracketedIds.length <= 4, `brackets=${L.bracketedIds}`)
        const before = m.savedCount
        ok("C2.selectOlives", L.selectById("olivabogyo") && L.activeCardWordId === "olivabogyo")
        L.saveCurrent()
        ok("C2.saveIncrements", m.savedCount === before + 1, `saved=${m.savedCount}`)
        for (const id of ["sonka", "sajt", "paradicsom"]) { L.selectById(id); L.saveCurrent() }
        ok("C2.savedFour", m.savedCount >= 4, `saved=${m.savedWords}`)
        // C3 Use it live
        m.selectMode("live")
        const plan: {[k: number]: string[]} = {1: ["jonapot", "szendvicset"], 2: ["sonka_sajt", "kerhetek_oliva", "done"], 3: ["tejes_kave"], 4: ["itt_eszem"], 5: ["nyolcas_asztal"], 6: ["koszonom"]}
        let tableDone = false, highlightSeen = false, chimeStep = false, lastStep = 0
        const t0 = getTime()
        const ev = this.createEvent("UpdateEvent")
        let nextAct = 0
        ev.bind(() => {
          if (getTime() < nextAct) return
          nextAct = getTime() + 0.6
          if (O.orderStep !== lastStep) { lastStep = O.orderStep; print(`PROBE step ${O.orderStep} ${O.orderStepName} line="${O.lastServerLine}"`) }
          if (O.orderStep === 2 && L.highlightIds.length > 0) highlightSeen = true
          if (O.orderStep === 5) chimeStep = true
          if (O.waitingFor === "table" && !tableDone) {
            tableDone = true
            O.onTable(8)
            const card = O.tableCards ? O.tableCards[8] : null
            ok("C3.tableTap8", O.lastTableTapped === 8 && O.chosenTable === 8)
            return
          }
          if (O.waitingFor === "user") {
            const want = plan[O.orderStep] || []
            for (const id of want) if (O.offeredPhraseIds.indexOf(id) >= 0 && O.answers.indexOf(id) < 0) { O.pickPhrase(id); return }
          }
          if (O.orderStep >= 7 || getTime() - t0 > 90) {
            ev.enabled = false
            ok("C3.orderComplete", O.orderStep === 7 && O.waitingFor === "done", `answers=${O.answers}`)
            ok("C3.ingredientHighlight", highlightSeen)
            ok("C3.tableStepReached", chimeStep && tableDone)
            // C4 Practice
            m.selectMode("practice")
            this.later(1.5, () => {
              ok("C4.boardOpen", P.active && P.pairsTotal === 4, `left=${P.leftIds} right=${P.rightIds}`)
              const wrong = P.leftIds[0] === P.rightIds[0] ? P.rightIds[1] : P.rightIds[0]
              P.pick("hu", P.leftIds[0]); P.pick("en", wrong === P.leftIds[0] ? P.rightIds[1] : wrong)
              ok("C4.wrongShowsTryAgain", P.wrongAttempts === 1)
              const ids = P.leftIds.slice()
              let i = 0
              const step = () => {
                if (i >= ids.length) {
                  this.later(3.0, () => {
                    ok("C4.complete", P.complete === true, `matched=${P.pairsMatched}/${P.pairsTotal}`)
                    // C5 mode switch keeps saved words
                    const sc = m.savedCount
                    m.toggleMenu(); m.selectMode("learn")
                    ok("C5.modeSwitchKeepsSaved", m.savedCount === sc && m.currentMode === "learn", `switches=${m.modeSwitches}`)
                    print(`PROBE SUMMARY ${R.filter((r) => r.indexOf("PASS") === 0).length}/${R.length} passed`)
                  })
                  return
                }
                P.pick("hu", ids[i]); P.pick("en", ids[i]); i++
                this.later(1.6, step)
              }
              this.later(0.5, step)
            })
          }
        })
      })
    })
  }

  private run(): void {
    const m = this.main
    const L = (m as any).learn, O = (m as any).order, P = (m as any).practice
    print(`PROBE stage=${this.stage} cam=${camPos()} look=${camLook()}`)
    if (this.stage === "setup") return
    if (this.stage.indexOf("view:") === 0) {
      // screenshot-only: "view:<target>[:<anchor>]" — stand the (fixed) preview head on <anchor>
      // and turn the café so <target> is straight ahead
      const parts = this.stage.split(":")
      ;(m as any).setupPanel.hide()
      this.later(2.6, () => {
        this.frame(parts[1], parts[2] || undefined)
        if (parts[3]) {
          // optional eye height (cm) — emulates looking down/up with the fixed preview head
          const ct = (m as any).cafe.getSceneObject().getTransform()
          const p = ct.getWorldPosition()
          ct.setWorldPosition(new vec3(p.x, camPos().y - Number(parts[3]), p.z))
        }
        if (parts[4]) {
          // optional head pitch (deg, + = look up): tilt the café around the eye instead
          const ct = (m as any).cafe.getSceneObject().getTransform()
          const c = camPos()
          const lk = camLook()
          const right = new vec3(-lk.z, 0, lk.x).normalize()
          const q = quat.angleAxis((-Number(parts[4]) * Math.PI) / 180, right)
          const rel = ct.getWorldPosition().sub(c)
          ct.setWorldPosition(c.add(q.multiplyVec3(rel)))
          ct.setWorldRotation(q.multiply(ct.getWorldRotation()))
        }
      })
      return
    }
    if (this.stage.indexOf("live:") === 0) {
      // screenshot-only: "live:<pitchDeg>[:<target>]" — Use it live from the start position, turned
      // toward <target> (default ServerSpot); the café is tilted about the eye to emulate head pitch
      // relative to the fixed preview head (+ = look up)
      const parts = this.stage.split(":")
      // optional 6th part: words to save first (comma list) so the filling phrases favour them
      if (parts[5]) for (const w of parts[5].split(",")) (m as any).session.save(w)
      m.selectMode("live")
      this.later(2.6, () => {
        this.frame(parts[2] || "ServerSpot")
        const ct = (m as any).cafe.getSceneObject().getTransform()
        const c = camPos()
        const lk = camLook()
        const right = new vec3(-lk.z, 0, lk.x).normalize()
        const q = quat.angleAxis((-Number(parts[1]) * Math.PI) / 180, right)
        const rel = ct.getWorldPosition().sub(c)
        ct.setWorldPosition(c.add(q.multiplyVec3(rel)))
        ct.setWorldRotation(q.multiply(ct.getWorldRotation()))
      })
      // the phrase row is laid on the counter tray when shown; re-lay it after the café was moved
      const relay = () => {
        if (!O.phraseTray) return
        const tt = O.phraseTray.getTransform(), pt = O.phrases.getSceneObject().getTransform()
        pt.setWorldPosition(tt.getWorldPosition())
        pt.setWorldRotation(tt.getWorldRotation())
        // the server caption is placed at its anchor when the line starts: re-place it and keep it up
        if (O.caption.isVisible) {
          placeFacingUser(O.caption.getSceneObject(), O.server.getCaptionAnchor(), true)
          O.captionHideAt = -1
        }
      }
      this.later(2.8, relay)
      this.later(4.5, relay)
      if (parts[3]) {
        // optional: answer with the first offered phrase until step <parts[3]> is reached
        const tick = () => {
          if (O.orderStepName === parts[3]) {
            relay()
            // show the picked (selected) state on one card
            if (parts[4]) this.later(1.5, () => print(`PROBE pick ${parts[4]} -> ${O.pickPhrase(parts[4])}`))
            return
          }
          if (O.waitingFor === "user" && O.offeredPhraseIds.length > 0) {
            const ids = O.offeredPhraseIds.filter((x: string) => x !== "done")
            const id = ids[ids.length - 1] || "done"
            print(`PROBE advance ${O.orderStepName} pick ${id} -> ${O.pickPhrase(id)}`)
          }
          this.later(1.2, tick)
        }
        this.later(3.2, tick)
      }
      this.later(parts[3] ? 25 : 7, () => {
        const hud = (m as any).hud
        print(`PROBE live step=${O.orderStep} ${O.orderStepName} offered=${O.offeredPhraseIds} hudCompact=${hud.isCompact} phrasePanel=${O.phrases.lastSizeCm}`)
      })
      return
    }
    if (this.stage.indexOf("wordcard") === 0 || this.stage === "hover") {
      const wid = this.stage.split(":")[1] || "olivabogyo"
      m.selectMode("learn")
      this.later(1.2, () => this.frame("Learnable_" + wid))
      this.later(4.5, () => {
        print(`PROBE select ${wid} -> ${L.selectById(wid)} card=${L.activeCardWordId} brackets=${L.bracketedIds}`)
        if (this.stage === "hover") this.later(0.6, () => {
          const card = (m as any).learn.wordCard
          const vis = (name: string) => {
            const so = this.find(card.getSceneObject(), name)
            const b = so ? (so.getComponent(Button.getTypeName()) as any) : null
            return b ? b.visual : null
          }
          const pv = vis("WordCardPlayButton"); if (pv) pv.setState("hovered")
          const sv = vis("WordCardPlaySentenceButton"); if (sv) sv.setState("triggered")
          print(`PROBE hover state set play=${!!pv} sentence=${!!sv}`)
        })
      })
      return
    }
    if (this.stage === "phrases" || this.stage === "phrases2") {
      // screenshot-only hack: turn the café so the counter is where the (moved) preview head looks
      const look = camLook()
      const yaw = Math.atan2(-look.x, -look.z)
      ;(m as any).cafe.getSceneObject().getTransform().setWorldRotation(quat.fromEulerAngles(0, yaw, 0))
      // the fixed preview head only looks ~13° down; lift the café so the counter tray sits where a user looking down at it would see it
      ;(m as any).cafe.eyeHeightCm = 138
      this.later(1.5, () => {
        const ct = (m as any).cafe.getSceneObject().getTransform()
        const p = ct.getWorldPosition()
        ct.setWorldPosition(new vec3(p.x, camPos().y - 138, p.z))
      })
      m.selectMode("live")
      this.later(5, () => {
        print(`PROBE order step=${O.orderStep} ${O.orderStepName} offered=${O.offeredPhraseIds}`)
        if (this.stage === "phrases2") {
          for (const id of ["jonapot", "szendvicset"]) if (O.offeredPhraseIds.indexOf(id) >= 0) print(`PROBE pick ${id} -> ${O.pickPhrase(id)}`)
          this.later(6, () => print(`PROBE order step=${O.orderStep} ${O.orderStepName} offered=${O.offeredPhraseIds} highlight=${L.highlightIds}`))
        }
      })
      return
    }
    if (this.stage === "menu") {
      // screenshot-only: the mode menu (after setup), with the Learn tile targeted
      m.selectMode("learn")
      this.later(1.0, () => m.toggleMenu())
      return
    }
    if (this.stage === "flow") {
      this.runFlow()
      return
    }
    if (this.stage === "practice") {
      for (const id of ["szalami", "paprika", "sajt", "uborka_friss"]) (m as any).session.save(id)
      m.selectMode("practice")
      this.later(3.0, () => {
        print(`PROBE practice active=${P.active} left=${P.leftIds}`)
        if (P.leftIds.length > 0) {
          const a = P.leftIds[0], b = P.leftIds[1]
          P.pick("hu", a); P.pick("en", a)
          this.later(2.0, () => { P.pick("hu", b); print(`PROBE practice matched ${a}, selected ${b}`) })
        }
      })
      return
    }
  }
}
